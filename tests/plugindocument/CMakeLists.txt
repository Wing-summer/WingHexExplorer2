cmake_minimum_required(VERSION 3.20)
project(PluginDocumentTests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17 CACHE STRING "C++ standard for document tests")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(WING_SOURCE_ROOT "../.." ABSOLUTE)
add_subdirectory("${WING_SOURCE_ROOT}/WingPlugin" WingPlugin)
add_executable(plugin_document_tests tst_plugindocument.cpp)
target_link_libraries(plugin_document_tests PRIVATE WingPlugin Qt6::Widgets Qt6::Test)
enable_testing()
add_test(NAME plugin_document_tests COMMAND plugin_document_tests)
set_tests_properties(plugin_document_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

add_test(NAME legacy_plugin_api COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_legacy_api.py")
