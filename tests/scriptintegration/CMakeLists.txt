cmake_minimum_required(VERSION 3.20)
project(ScriptIntegrationTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17 CACHE STRING "C++ standard")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network Test)
find_package(OpenSSL REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(OPENSSL_CLI NAMES openssl REQUIRED)
get_filename_component(HOST "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(script_integration_tests tst_scriptintegration.cpp
    "${HOST}/src/class/security/wingcert.cpp"
    "${HOST}/src/class/file/fileaccesscheck.cpp")
target_include_directories(script_integration_tests PRIVATE "${HOST}/src")
target_link_libraries(script_integration_tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test OpenSSL::Crypto)
if(WIN32)
    target_link_libraries(script_integration_tests PRIVATE advapi32)
endif()
enable_testing()
add_test(NAME script_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run.py" "$<TARGET_FILE:script_integration_tests>" "${OPENSSL_CLI}")
