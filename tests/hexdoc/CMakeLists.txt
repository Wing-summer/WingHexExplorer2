cmake_minimum_required(VERSION 3.20)
project(HexdocTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17 CACHE STRING "C++ standard for the runtime fixture")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
get_filename_component(WING_ROOT "../.." ABSOLUTE)
set(HEX_BINDINGS "${WING_ROOT}/build/generated/WingLuauBind" CACHE PATH "Generated hexdoc bindings")
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_WEB OFF CACHE BOOL "" FORCE)
set(LUAU_WERROR OFF CACHE BOOL "" FORCE)
add_subdirectory("${WING_ROOT}/3rdparty/luau" luau EXCLUDE_FROM_ALL)
add_executable(hexdoc_runtime runtime.cpp "${WING_ROOT}/src/luau/wingluaurequire.cpp")
target_include_directories(hexdoc_runtime PRIVATE "${WING_ROOT}/src" "${WING_ROOT}/3rdparty")
target_link_libraries(hexdoc_runtime PRIVATE Qt6::Core Luau.Require Luau.Compiler Luau.VM)
target_compile_definitions(hexdoc_runtime PRIVATE WING_SOURCE="${WING_ROOT}" HEX_BINDINGS="${HEX_BINDINGS}")
enable_testing()
add_test(NAME hexdoc_runtime COMMAND hexdoc_runtime)
